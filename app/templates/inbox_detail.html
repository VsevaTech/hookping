How to sign a request

Sign <timestamp>.<raw body> with HMAC-SHA256 and send {{ inbox.timestamp_header }}: <unix seconds> and {{ inbox.signature_header }}: sha256=<hex>. Sign the exact bytes you send. Several comma-separated signatures are accepted, which helps while rotating secrets. 

SECRET='whsec_…'
BODY='{"name": "Ivan", "plan": "Pro", "amount": 250}'
TS=$(date +%s)
SIG=$(printf '%s.%s' "$TS" "$BODY" | openssl dgst -sha256 -hmac "$SECRET" | sed 's/^.* //')

curl -X POST {{ url }} \
  -H "Content-Type: application/json" \
  -H "{{ inbox.timestamp_header }}: $TS" \
  -H "{{ inbox.signature_header }}: sha256=$SIG" \
  --data-binary "$BODY"
import hashlib, hmac, json, time, httpx

body = json.dumps({"name": "Ivan", "plan": "Pro"}).encode()
ts = str(int(time.time()))
sig = hmac.new(SECRET.encode(), ts.encode() + b"." + body, hashlib.sha256).hexdigest()
httpx.post("{{ url }}", content=body, headers={
    "Content-Type": "application/json",
    "{{ inbox.timestamp_header }}": ts,
    "{{ inbox.signature_header }}": f"sha256={sig}",
})
